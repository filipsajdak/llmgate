function(llmgate_module name)
  cmake_parse_arguments(ARG "" "" "INTERFACE;SOURCES;LIBS" ${ARGN})
  add_library(${name})
  target_sources(${name} PUBLIC FILE_SET CXX_MODULES BASE_DIRS src FILES ${ARG_INTERFACE})
  if(ARG_SOURCES)
    target_sources(${name} PRIVATE ${ARG_SOURCES})
  endif()
  target_link_libraries(${name} PUBLIC llmgate_flags llmgate_contracts ${ARG_LIBS})
endfunction()

function(llmgate_constexpr_corpus name property directory)
  file(GLOB corpus CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/tests/fuzz/${directory}/corpus/*)
  set(body "import llmgate.properties;\nimport std;\n\nnamespace {\n")
  set(index 0)
  foreach(file IN LISTS corpus)
    string(APPEND body "constexpr char file_${index}[] = {\n#embed \"${file}\"\n};\n")
    string(APPEND body "static_assert(llmgate::properties::${property}(std::string_view{file_${index}, sizeof file_${index}}), \"${file}\");\n")
    math(EXPR index "${index}+1")
  endforeach()
  string(APPEND body "}  // namespace\n\nint main() {}\n")
  file(GENERATE OUTPUT ${CMAKE_BINARY_DIR}/constexpr/${name}.cpp CONTENT "${body}")
  add_executable(constexpr_${name} ${CMAKE_BINARY_DIR}/constexpr/${name}.cpp)
  target_compile_options(constexpr_${name} PRIVATE -fno-sanitize=all)
  target_link_libraries(constexpr_${name} PRIVATE llmgate_properties llmgate_contracts)
  add_test(NAME constexpr_${name} COMMAND constexpr_${name})
  set_tests_properties(constexpr_${name} PROPERTIES LABELS constexpr)
endfunction()

  target_compile_definitions(constexpr_exhaustive_${name} PRIVATE LLMGATE_CHECK=${check} ${exhaustive_lengths})
